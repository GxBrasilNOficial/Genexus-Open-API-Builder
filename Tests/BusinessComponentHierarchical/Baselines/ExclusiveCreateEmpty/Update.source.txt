&RestStatusCode = 200
If &UpdateRequest.FirmId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"FirmId")
EndIf
If Len(&UpdateRequest.FirmName) > 40
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"FirmName")
EndIf
If &RestStatusCode = 200
    &Firm.Load(&FirmId)
    If &Firm.Success()
        &Firm.FirmId = &UpdateRequest.FirmId
        &Firm.FirmName = &UpdateRequest.FirmName
        If &UpdateRequest.ExclusiveReplace
            &Firm.Exclusive.Clear()
            For &Update_Exclusive in &UpdateRequest.Exclusive
                &Bc_Exclusive = new()
                &Firm.Exclusive.Add(&Bc_Exclusive)
            EndFor
        EndIf
        &Firm.Save()
        If &Firm.Success()
            Commit
            &Firm.Load(&FirmId)
            &UpdateResponse = new()
            &UpdateResponse.FirmId = &Firm.FirmId
            &UpdateResponse.FirmName = &Firm.FirmName
            For &Bc_Exclusive in &Firm.Exclusive
                &Get_Exclusive = new()
                &Get_Exclusive.FirmId = &Bc_Exclusive.FirmId
                &UpdateResponse.Exclusive.Add(&Get_Exclusive)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &Firm.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"Firm was not found."
    EndIf
EndIf
