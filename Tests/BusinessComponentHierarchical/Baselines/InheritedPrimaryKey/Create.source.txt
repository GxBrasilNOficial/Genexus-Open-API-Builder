&RestStatusCode = 201
&Header = new()
For &Create_Level1 in &CreateRequest.Level1
    &Bc_Level1 = new()
    &Bc_Level1.LineId = &Create_Level1.LineId
    &Bc_Level1.LineText = &Create_Level1.LineText
    &Header.<unnamed>.Add(&Bc_Level1)
EndFor
&Header.Save()
If &Header.Success()
    Commit
    &Header.Load(&Header.HeaderId)
    &CreateResponse = new()
    &CreateResponse.HeaderId = &Header.HeaderId
    For &Bc_Level1 in &Header.<unnamed>
        &Get_Level1 = new()
        &Get_Level1.LineId = &Bc_Level1.LineId
        &Get_Level1.LineText = &Bc_Level1.LineText
        &Get_Level1.HeaderId = &Bc_Level1.HeaderId
        &CreateResponse.Level1.Add(&Get_Level1)
    EndFor
    &LocationUrl = !"/header"
    &LocationUrl = &LocationUrl + !"/" + &Header.HeaderId.ToString().Trim()
    &HttpResponse.AddHeader(!"Location", &LocationUrl)
    &RestStatusCode = 201
Else
    &RestStatusCode = 422
    &ErrorResponse = new()
    &ErrorResponse.Code = !"validation_error"
    &Messages = &Header.GetMessages()
    &ConcatenatedMessage = !""
    For &i = 1 To &Messages.Count
        &Message = &Messages.Item(&i)
        If &Message.Type = MessageTypes.Error
            &ErrorItem = new()
            If &Message.Id.IsEmpty()
                &ErrorItem.Code = !"business_rule"
            Else
                &ErrorItem.Code = &Message.Id
            EndIf
            &ErrorItem.Message = &Message.Description
            &ErrorResponse.Messages.Add(&ErrorItem)
            If &ConcatenatedMessage.IsEmpty()
                &ConcatenatedMessage = &Message.Description
            Else
                &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
            EndIf
        EndIf
    EndFor
    If &ConcatenatedMessage.IsEmpty()
        &ErrorResponse.Message = !"Business rules rejected the request."
    Else
        If Len(&ConcatenatedMessage) > 2048
            &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
        Else
            &ErrorResponse.Message = &ConcatenatedMessage
        EndIf
    EndIf
EndIf
