&RestStatusCode = 200
If &UpdateRequest.HeaderId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"HeaderId")
EndIf
If &RestStatusCode = 200
    &Header.Load(&HeaderId)
    If &Header.Success()
        &Header.HeaderId = &UpdateRequest.HeaderId
        If &UpdateRequest.LineReplace
            &Idx_Line = &Header.Line.Count
            Do While &Idx_Line >= 1
                &Bc_Line = &Header.Line.Item(&Idx_Line)
                &Found_Line = False
                For &Update_Line in &UpdateRequest.Line
                    If &Update_Line.HeaderId = &Bc_Line.HeaderId and &Update_Line.LineId = &Bc_Line.LineId
                        &Found_Line = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Line
                    &Header.Line.Remove(&Idx_Line)
                EndIf
                &Idx_Line = &Idx_Line - 1
            EndDo
            For &Update_Line in &UpdateRequest.Line
                &Found_Line = False
                For &Bc_Line in &Header.Line
                    If &Update_Line.HeaderId = &Bc_Line.HeaderId and &Update_Line.LineId = &Bc_Line.LineId
                        &Found_Line = True
                        &Bc_Line.LineId = &Update_Line.LineId
                        &Bc_Line.LineText = &Update_Line.LineText
                        &Bc_Line.HeaderId = &Update_Line.HeaderId
                        Exit
                    EndIf
                EndFor
                If not &Found_Line
                    &Bc_Line = new()
                        &Bc_Line.LineId = &Update_Line.LineId
                        &Bc_Line.LineText = &Update_Line.LineText
                        &Bc_Line.HeaderId = &Update_Line.HeaderId
                    &Header.Line.Add(&Bc_Line)
                EndIf
            EndFor
        EndIf
        &Header.Save()
        If &Header.Success()
            Commit
            &Header.Load(&HeaderId)
            &UpdateResponse = new()
            &UpdateResponse.HeaderId = &Header.HeaderId
            For &Bc_Line in &Header.Line
                &Get_Line = new()
                &Get_Line.LineId = &Bc_Line.LineId
                &Get_Line.LineText = &Bc_Line.LineText
                &Get_Line.HeaderId = &Bc_Line.HeaderId
                &UpdateResponse.Line.Add(&Get_Line)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &Header.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"Header was not found."
    EndIf
EndIf
