&RestStatusCode = 200
If &UpdateRequest.HeaderId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"HeaderId")
EndIf
If &RestStatusCode = 200
    &Header.Load(&HeaderId)
    If &Header.Success()
        &Header.HeaderId = &UpdateRequest.HeaderId
        If &UpdateRequest.Level1Replace
            &Idx_Level1 = &Header.<unnamed>.Count
            Do While &Idx_Level1 >= 1
                &Bc_Level1 = &Header.<unnamed>.Item(&Idx_Level1)
                &Found_Level1 = False
                For &Update_Level1 in &UpdateRequest.Level1
                    If &Update_Level1.HeaderId = &Bc_Level1.HeaderId and &Update_Level1.LineId = &Bc_Level1.LineId
                        &Found_Level1 = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Level1
                    &Header.<unnamed>.Remove(&Idx_Level1)
                EndIf
                &Idx_Level1 = &Idx_Level1 - 1
            EndDo
            For &Update_Level1 in &UpdateRequest.Level1
                &Found_Level1 = False
                For &Bc_Level1 in &Header.<unnamed>
                    If &Update_Level1.HeaderId = &Bc_Level1.HeaderId and &Update_Level1.LineId = &Bc_Level1.LineId
                        &Found_Level1 = True
                        &Bc_Level1.LineId = &Update_Level1.LineId
                        &Bc_Level1.LineText = &Update_Level1.LineText
                        &Bc_Level1.HeaderId = &Update_Level1.HeaderId
                        Exit
                    EndIf
                EndFor
                If not &Found_Level1
                    &Bc_Level1 = new()
                        &Bc_Level1.LineId = &Update_Level1.LineId
                        &Bc_Level1.LineText = &Update_Level1.LineText
                        &Bc_Level1.HeaderId = &Update_Level1.HeaderId
                    &Header.<unnamed>.Add(&Bc_Level1)
                EndIf
            EndFor
        EndIf
        &Header.Save()
        If &Header.Success()
            Commit
            &Header.Load(&HeaderId)
            &UpdateResponse = new()
            &UpdateResponse.HeaderId = &Header.HeaderId
            For &Bc_Level1 in &Header.<unnamed>
                &Get_Level1 = new()
                &Get_Level1.LineId = &Bc_Level1.LineId
                &Get_Level1.LineText = &Bc_Level1.LineText
                &Get_Level1.HeaderId = &Bc_Level1.HeaderId
                &UpdateResponse.Level1.Add(&Get_Level1)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &Header.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"Header was not found."
    EndIf
EndIf
