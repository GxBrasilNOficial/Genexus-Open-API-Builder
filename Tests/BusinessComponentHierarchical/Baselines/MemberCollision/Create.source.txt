&RestStatusCode = 201
&PayloadErrorCode = !""
&PayloadErrorMessage = !""
If Len(&CreateRequest.Notes) > 40
    &RestStatusCode = 400
    If &PayloadErrorCode.IsEmpty()
        &PayloadErrorCode = !"attribute_limit_exceeded"
        &ErrorResponse = new()
    EndIf
    &PayloadErrorItem = new()
    &PayloadErrorItem.Code = !"attribute_limit_exceeded"
    &PayloadErrorItem.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"Notes")
    &ErrorResponse.Messages.Add(&PayloadErrorItem)
    If &PayloadErrorMessage.IsEmpty()
        &PayloadErrorMessage = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"Notes")
    Else
        &PayloadErrorMessage = &PayloadErrorMessage + !" | " + Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"Notes")
    EndIf
EndIf
If &RestStatusCode = 201
    &CollisionDoc = new()
    &CollisionDoc.Notes = &CreateRequest.Notes
    For &Create_Notes in &CreateRequest.Notes1
        &Bc_Notes = new()
        &Bc_Notes.NoteId = &Create_Notes.NoteId
        &Bc_Notes.NoteText = &Create_Notes.NoteText
        &CollisionDoc.Notes.Add(&Bc_Notes)
    EndFor
    &CollisionDoc.Save()
    If &CollisionDoc.Success()
        Commit
        &CollisionDoc.Load(&CollisionDoc.DocId)
        &CreateResponse = new()
        &CreateResponse.DocId = &CollisionDoc.DocId
        &CreateResponse.Notes = &CollisionDoc.Notes
        For &Bc_Notes in &CollisionDoc.Notes
            &Get_Notes = new()
            &Get_Notes.NoteId = &Bc_Notes.NoteId
            &Get_Notes.NoteText = &Bc_Notes.NoteText
            &CreateResponse.Notes1.Add(&Get_Notes)
        EndFor
        &LocationUrl = !"/collisiondoc"
        &LocationUrl = &LocationUrl + !"/" + &CollisionDoc.DocId.ToString().Trim()
        &HttpResponse.AddHeader(!"Location", &LocationUrl)
        &RestStatusCode = 201
    Else
        &RestStatusCode = 422
        &ErrorResponse = new()
        &ErrorResponse.Code = !"validation_error"
        &Messages = &CollisionDoc.GetMessages()
        &ConcatenatedMessage = !""
        For &i = 1 To &Messages.Count
            &Message = &Messages.Item(&i)
            If &Message.Type = MessageTypes.Error
                &ErrorItem = new()
                If &Message.Id.IsEmpty()
                    &ErrorItem.Code = !"business_rule"
                Else
                    &ErrorItem.Code = &Message.Id
                EndIf
                &ErrorItem.Message = &Message.Description
                &ErrorResponse.Messages.Add(&ErrorItem)
                If &ConcatenatedMessage.IsEmpty()
                    &ConcatenatedMessage = &Message.Description
                Else
                    &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                EndIf
            EndIf
        EndFor
        If &ConcatenatedMessage.IsEmpty()
            &ErrorResponse.Message = !"Business rules rejected the request."
        Else
            If Len(&ConcatenatedMessage) > 2048
                &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
            Else
                &ErrorResponse.Message = &ConcatenatedMessage
            EndIf
        EndIf
    EndIf
If not &PayloadErrorMessage.IsEmpty()
    &ErrorResponse.Code = &PayloadErrorCode
    &ErrorResponse.Message = &PayloadErrorMessage
EndIf
EndIf
