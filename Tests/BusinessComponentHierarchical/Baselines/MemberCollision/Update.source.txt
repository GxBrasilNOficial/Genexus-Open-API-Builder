&RestStatusCode = 200
If &UpdateRequest.DocId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"DocId")
EndIf
If Len(&UpdateRequest.Notes) > 40
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"Notes")
EndIf
If &RestStatusCode = 200
    &CollisionDoc.Load(&DocId)
    If &CollisionDoc.Success()
        &CollisionDoc.DocId = &UpdateRequest.DocId
        &CollisionDoc.Notes = &UpdateRequest.Notes
        If &UpdateRequest.Notes1Replace
            &Idx_Notes = &CollisionDoc.Notes.Count
            Do While &Idx_Notes >= 1
                &Bc_Notes = &CollisionDoc.Notes.Item(&Idx_Notes)
                &Found_Notes = False
                For &Update_Notes in &UpdateRequest.Notes1
                    If &Update_Notes.NoteId = &Bc_Notes.NoteId
                        &Found_Notes = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Notes
                    &CollisionDoc.Notes.Remove(&Idx_Notes)
                EndIf
                &Idx_Notes = &Idx_Notes - 1
            EndDo
            For &Update_Notes in &UpdateRequest.Notes1
                &Found_Notes = False
                For &Bc_Notes in &CollisionDoc.Notes
                    If &Update_Notes.NoteId = &Bc_Notes.NoteId
                        &Found_Notes = True
                        &Bc_Notes.NoteId = &Update_Notes.NoteId
                        &Bc_Notes.NoteText = &Update_Notes.NoteText
                        Exit
                    EndIf
                EndFor
                If not &Found_Notes
                    &Bc_Notes = new()
                        &Bc_Notes.NoteId = &Update_Notes.NoteId
                        &Bc_Notes.NoteText = &Update_Notes.NoteText
                    &CollisionDoc.Notes.Add(&Bc_Notes)
                EndIf
            EndFor
        EndIf
        &CollisionDoc.Save()
        If &CollisionDoc.Success()
            Commit
            &CollisionDoc.Load(&DocId)
            &UpdateResponse = new()
            &UpdateResponse.DocId = &CollisionDoc.DocId
            &UpdateResponse.Notes = &CollisionDoc.Notes
            For &Bc_Notes in &CollisionDoc.Notes
                &Get_Notes = new()
                &Get_Notes.NoteId = &Bc_Notes.NoteId
                &Get_Notes.NoteText = &Bc_Notes.NoteText
                &UpdateResponse.Notes1.Add(&Get_Notes)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &CollisionDoc.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"CollisionDoc was not found."
    EndIf
EndIf
