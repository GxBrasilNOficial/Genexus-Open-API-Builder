&RestStatusCode = 200
If &UpdateRequest.OrderId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"OrderId")
EndIf
If Len(&UpdateRequest.OrderDesc) > 40
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"OrderDesc")
EndIf
If &RestStatusCode = 200
    &Order.Load(&OrderId)
    If &Order.Success()
        &Order.OrderId = &UpdateRequest.OrderId
        &Order.OrderDesc = &UpdateRequest.OrderDesc
        If &UpdateRequest.LinesReplace
            &Idx_Lines = &Order.Lines.Count
            Do While &Idx_Lines >= 1
                &Bc_Lines = &Order.Lines.Item(&Idx_Lines)
                &Found_Lines = False
                For &Update_Lines in &UpdateRequest.Lines
                    If &Update_Lines.LineId = &Bc_Lines.LineId
                        &Found_Lines = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Lines
                    &Order.Lines.Remove(&Idx_Lines)
                EndIf
                &Idx_Lines = &Idx_Lines - 1
            EndDo
            For &Update_Lines in &UpdateRequest.Lines
                &Found_Lines = False
                For &Bc_Lines in &Order.Lines
                    If &Update_Lines.LineId = &Bc_Lines.LineId
                        &Found_Lines = True
                        &Bc_Lines.LineId = &Update_Lines.LineId
                        &Bc_Lines.LineQty = &Update_Lines.LineQty
                        Exit
                    EndIf
                EndFor
                If not &Found_Lines
                    &Bc_Lines = new()
                        &Bc_Lines.LineId = &Update_Lines.LineId
                        &Bc_Lines.LineQty = &Update_Lines.LineQty
                    &Order.Lines.Add(&Bc_Lines)
                EndIf
            EndFor
        EndIf
        &Order.Save()
        If &Order.Success()
            Commit
            &Order.Load(&OrderId)
            &UpdateResponse = new()
            &UpdateResponse.OrderId = &Order.OrderId
            &UpdateResponse.OrderDesc = &Order.OrderDesc
            For &Bc_Lines in &Order.Lines
                &Get_Lines = new()
                &Get_Lines.LineId = &Bc_Lines.LineId
                &Get_Lines.LineQty = &Bc_Lines.LineQty
                &Get_Lines.LineTotal = &Bc_Lines.LineTotal
                &Get_Lines.LineStamp = &Bc_Lines.LineStamp
                &UpdateResponse.Lines.Add(&Get_Lines)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &Order.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"Order was not found."
    EndIf
EndIf
