&RestStatusCode = 200
If &UpdateRequest.DocId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"DocId")
EndIf
If &RestStatusCode = 200
    &Document.Load(&DocId)
    If &Document.Success()
        &Document.DocId = &UpdateRequest.DocId
        If &UpdateRequest.NotesReplace
            &Idx_Notes = &Document.Notes.Count
            Do While &Idx_Notes >= 1
                &Bc_Notes = &Document.Notes.Item(&Idx_Notes)
                &Found_Notes = False
                For &Update_Notes in &UpdateRequest.Notes
                    If &Update_Notes.NoteId = &Bc_Notes.NoteId
                        &Found_Notes = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Notes
                    &Document.Notes.Remove(&Idx_Notes)
                EndIf
                &Idx_Notes = &Idx_Notes - 1
            EndDo
            For &Update_Notes in &UpdateRequest.Notes
                &Found_Notes = False
                For &Bc_Notes in &Document.Notes
                    If &Update_Notes.NoteId = &Bc_Notes.NoteId
                        &Found_Notes = True
                        &Bc_Notes.NoteId = &Update_Notes.NoteId
                        &Bc_Notes.NoteText = &Update_Notes.NoteText
                        Exit
                    EndIf
                EndFor
                If not &Found_Notes
                    &Bc_Notes = new()
                        &Bc_Notes.NoteId = &Update_Notes.NoteId
                        &Bc_Notes.NoteText = &Update_Notes.NoteText
                    &Document.Notes.Add(&Bc_Notes)
                EndIf
            EndFor
        EndIf
        If &UpdateRequest.TagsReplace
            &Idx_Tags = &Document.Tags.Count
            Do While &Idx_Tags >= 1
                &Bc_Tags = &Document.Tags.Item(&Idx_Tags)
                &Found_Tags = False
                For &Update_Tags in &UpdateRequest.Tags
                    If &Update_Tags.TagId = &Bc_Tags.TagId
                        &Found_Tags = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Tags
                    &Document.Tags.Remove(&Idx_Tags)
                EndIf
                &Idx_Tags = &Idx_Tags - 1
            EndDo
            For &Update_Tags in &UpdateRequest.Tags
                &Found_Tags = False
                For &Bc_Tags in &Document.Tags
                    If &Update_Tags.TagId = &Bc_Tags.TagId
                        &Found_Tags = True
                        &Bc_Tags.TagId = &Update_Tags.TagId
                        Exit
                    EndIf
                EndFor
                If not &Found_Tags
                    &Bc_Tags = new()
                        &Bc_Tags.TagId = &Update_Tags.TagId
                    &Document.Tags.Add(&Bc_Tags)
                EndIf
            EndFor
        EndIf
        &Document.Save()
        If &Document.Success()
            Commit
            &Document.Load(&DocId)
            &UpdateResponse = new()
            &UpdateResponse.DocId = &Document.DocId
            For &Bc_Notes in &Document.Notes
                &Get_Notes = new()
                &Get_Notes.NoteId = &Bc_Notes.NoteId
                &Get_Notes.NoteText = &Bc_Notes.NoteText
                &UpdateResponse.Notes.Add(&Get_Notes)
            EndFor
            For &Bc_Tags in &Document.Tags
                &Get_Tags = new()
                &Get_Tags.TagId = &Bc_Tags.TagId
                &Get_Tags.TagCode = &Bc_Tags.TagCode
                &UpdateResponse.Tags.Add(&Get_Tags)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &Document.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"Document was not found."
    EndIf
EndIf
