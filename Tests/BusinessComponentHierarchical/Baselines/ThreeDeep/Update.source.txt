&RestStatusCode = 200
If &UpdateRequest.DayId > 99999999
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"DayId")
EndIf
If &RestStatusCode = 200
    &Day.Load(&DayId)
    If &Day.Success()
        &Day.DayId = &UpdateRequest.DayId
        If &UpdateRequest.ShiftReplace
            &Idx_Shift = &Day.Shift.Count
            Do While &Idx_Shift >= 1
                &Bc_Shift = &Day.Shift.Item(&Idx_Shift)
                &Found_Shift = False
                For &Update_Shift in &UpdateRequest.Shift
                    If &Update_Shift.ShiftId = &Bc_Shift.ShiftId
                        &Found_Shift = True
                        Exit
                    EndIf
                EndFor
                If not &Found_Shift
                    &Day.Shift.Remove(&Idx_Shift)
                EndIf
                &Idx_Shift = &Idx_Shift - 1
            EndDo
            For &Update_Shift in &UpdateRequest.Shift
                &Found_Shift = False
                For &Bc_Shift in &Day.Shift
                    If &Update_Shift.ShiftId = &Bc_Shift.ShiftId
                        &Found_Shift = True
                        &Bc_Shift.ShiftId = &Update_Shift.ShiftId
                        &Bc_Shift.ShiftName = &Update_Shift.ShiftName
                        If &Update_Shift.WorkerReplace
                            &Bc_Shift.Worker.Clear()
                            For &Update_Shift_Worker in &Update_Shift.Worker
                                &Bc_Shift_Worker = new()
                                &Bc_Shift_Worker.WorkerName = &Update_Shift_Worker.WorkerName
                                &Bc_Shift.Worker.Add(&Bc_Shift_Worker)
                            EndFor
                        EndIf
                        Exit
                    EndIf
                EndFor
                If not &Found_Shift
                    &Bc_Shift = new()
                        &Bc_Shift.ShiftId = &Update_Shift.ShiftId
                        &Bc_Shift.ShiftName = &Update_Shift.ShiftName
                    For &Update_Shift_Worker in &Update_Shift.Worker
                        &Bc_Shift_Worker = new()
                        &Bc_Shift_Worker.WorkerName = &Update_Shift_Worker.WorkerName
                        &Bc_Shift.Worker.Add(&Bc_Shift_Worker)
                    EndFor
                    &Day.Shift.Add(&Bc_Shift)
                EndIf
            EndFor
        EndIf
        &Day.Save()
        If &Day.Success()
            Commit
            &Day.Load(&DayId)
            &UpdateResponse = new()
            &UpdateResponse.DayId = &Day.DayId
            For &Bc_Shift in &Day.Shift
                &Get_Shift = new()
                &Get_Shift.ShiftId = &Bc_Shift.ShiftId
                &Get_Shift.ShiftName = &Bc_Shift.ShiftName
                For &Bc_Shift_Worker in &Bc_Shift.Worker
                    &Get_Shift_Worker = new()
                    &Get_Shift_Worker.WorkerId = &Bc_Shift_Worker.WorkerId
                    &Get_Shift_Worker.WorkerName = &Bc_Shift_Worker.WorkerName
                    &Get_Shift_Worker.WorkerScore = &Bc_Shift_Worker.WorkerScore
                    &Get_Shift.Worker.Add(&Get_Shift_Worker)
                EndFor
                &UpdateResponse.Shift.Add(&Get_Shift)
            EndFor
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &Day.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"Day was not found."
    EndIf
EndIf
