&RestStatusCode = 200
&RequestJsonHasRequiredMembers = True
&MissingRequiredFields = !""
&EmptyUpdateRequest = new()
If &UpdateRequest.FlatCompositeKeyDescription = &EmptyUpdateRequest.FlatCompositeKeyDescription
    &RequestJsonHasRequiredMembers = False
    If &MissingRequiredFields.IsEmpty()
        &MissingRequiredFields = !"FlatCompositeKeyDescription"
    Else
        &MissingRequiredFields = &MissingRequiredFields + !", FlatCompositeKeyDescription"
    EndIf
EndIf
If not &RequestJsonHasRequiredMembers
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"invalid_request"
    &ErrorResponse.Message = Format(!"Required JSON member(s) missing or empty: %1.", &MissingRequiredFields.Trim())
EndIf
If Len(&UpdateRequest.FlatCompositeKeyDescription) > 100
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"FlatCompositeKeyDescription")
EndIf
If &RestStatusCode = 200
    &FlatCompositeKey.Load(&FlatCompositeKeyCompanyId, &FlatCompositeKeyItemId, &FlatCompositeKeyCode)
    If &FlatCompositeKey.Success()
        &FlatCompositeKey.FlatCompositeKeyDescription = &UpdateRequest.FlatCompositeKeyDescription
        &FlatCompositeKey.Save()
        If &FlatCompositeKey.Success()
            Commit
            &FlatCompositeKey.Load(&FlatCompositeKeyCompanyId, &FlatCompositeKeyItemId, &FlatCompositeKeyCode)
            &UpdateResponse = new()
            &UpdateResponse.FlatCompositeKeyCompanyId = &FlatCompositeKey.FlatCompositeKeyCompanyId
            &UpdateResponse.FlatCompositeKeyItemId = &FlatCompositeKey.FlatCompositeKeyItemId
            &UpdateResponse.FlatCompositeKeyCode = &FlatCompositeKey.FlatCompositeKeyCode
            &UpdateResponse.FlatCompositeKeyDescription = &FlatCompositeKey.FlatCompositeKeyDescription
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &FlatCompositeKey.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"FlatCompositeKey was not found."
    EndIf
EndIf
