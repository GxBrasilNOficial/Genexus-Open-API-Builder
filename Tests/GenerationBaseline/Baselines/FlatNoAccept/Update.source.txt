&RestStatusCode = 200
&RequestJsonHasRequiredMembers = True
&MissingRequiredFields = !""
&EmptyUpdateRequest = new()
If &UpdateRequest.FlatNoAcceptTitle = &EmptyUpdateRequest.FlatNoAcceptTitle
    &RequestJsonHasRequiredMembers = False
    If &MissingRequiredFields.IsEmpty()
        &MissingRequiredFields = !"FlatNoAcceptTitle"
    Else
        &MissingRequiredFields = &MissingRequiredFields + !", FlatNoAcceptTitle"
    EndIf
EndIf
If not &RequestJsonHasRequiredMembers
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"invalid_request"
    &ErrorResponse.Message = Format(!"Required JSON member(s) missing or empty: %1.", &MissingRequiredFields.Trim())
EndIf
If Len(&UpdateRequest.FlatNoAcceptTitle) > 80
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"FlatNoAcceptTitle")
EndIf
If &RestStatusCode = 200
    &FlatNoAccept.Load(&FlatNoAcceptId)
    If &FlatNoAccept.Success()
        &FlatNoAccept.FlatNoAcceptTitle = &UpdateRequest.FlatNoAcceptTitle
        &FlatNoAccept.Save()
        If &FlatNoAccept.Success()
            Commit
            &FlatNoAccept.Load(&FlatNoAcceptId)
            &UpdateResponse = new()
            &UpdateResponse.FlatNoAcceptId = &FlatNoAccept.FlatNoAcceptId
            &UpdateResponse.FlatNoAcceptTitle = &FlatNoAccept.FlatNoAcceptTitle
            &UpdateResponse.FlatNoAcceptAddedDate = &FlatNoAccept.FlatNoAcceptAddedDate
            &RestStatusCode = 200
        Else
            &RestStatusCode = 422
            &ErrorResponse = new()
            &ErrorResponse.Code = !"validation_error"
            &Messages = &FlatNoAccept.GetMessages()
            &ConcatenatedMessage = !""
            For &i = 1 To &Messages.Count
                &Message = &Messages.Item(&i)
                If &Message.Type = MessageTypes.Error
                    &ErrorItem = new()
                    If &Message.Id.IsEmpty()
                        &ErrorItem.Code = !"business_rule"
                    Else
                        &ErrorItem.Code = &Message.Id
                    EndIf
                    &ErrorItem.Message = &Message.Description
                    &ErrorResponse.Messages.Add(&ErrorItem)
                    If &ConcatenatedMessage.IsEmpty()
                        &ConcatenatedMessage = &Message.Description
                    Else
                        &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                    EndIf
                EndIf
            EndFor
            If &ConcatenatedMessage.IsEmpty()
                &ErrorResponse.Message = !"Business rules rejected the request."
            Else
                If Len(&ConcatenatedMessage) > 2048
                    &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
                Else
                    &ErrorResponse.Message = &ConcatenatedMessage
                EndIf
            EndIf
        EndIf
    Else
        &RestStatusCode = 404
        &ErrorResponse = new()
        &ErrorResponse.Code = !"not_found"
        &ErrorResponse.Message = !"FlatNoAccept was not found."
    EndIf
EndIf
