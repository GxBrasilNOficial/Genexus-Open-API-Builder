&RestStatusCode = 201
If Len(&CreateRequest.FlatSimpleKeyName) > 60
    &RestStatusCode = 400
    &ErrorResponse = new()
    &ErrorResponse.Code = !"attribute_limit_exceeded"
    &ErrorResponse.Message = Format(!"Value of field %1 exceeds the limit defined by its attribute.", !"FlatSimpleKeyName")
EndIf
If &RestStatusCode = 201
    &FlatSimpleKey = new()
    &FlatSimpleKey.FlatSimpleKeyName = &CreateRequest.FlatSimpleKeyName
    &FlatSimpleKey.FlatSimpleKeyActive = &CreateRequest.FlatSimpleKeyActive
    &FlatSimpleKey.Save()
    If &FlatSimpleKey.Success()
        Commit
        &FlatSimpleKey.Load(&FlatSimpleKey.FlatSimpleKeyId)
        &CreateResponse = new()
        &CreateResponse.FlatSimpleKeyId = &FlatSimpleKey.FlatSimpleKeyId
        &CreateResponse.FlatSimpleKeyName = &FlatSimpleKey.FlatSimpleKeyName
        &CreateResponse.FlatSimpleKeyActive = &FlatSimpleKey.FlatSimpleKeyActive
        &LocationUrl = !"/flatsimplekey"
        &LocationUrl = &LocationUrl + !"/" + &FlatSimpleKey.FlatSimpleKeyId.ToString().Trim()
        &HttpResponse.AddHeader(!"Location", &LocationUrl)
        &RestStatusCode = 201
    Else
        &RestStatusCode = 422
        &ErrorResponse = new()
        &ErrorResponse.Code = !"validation_error"
        &Messages = &FlatSimpleKey.GetMessages()
        &ConcatenatedMessage = !""
        For &i = 1 To &Messages.Count
            &Message = &Messages.Item(&i)
            If &Message.Type = MessageTypes.Error
                &ErrorItem = new()
                If &Message.Id.IsEmpty()
                    &ErrorItem.Code = !"business_rule"
                Else
                    &ErrorItem.Code = &Message.Id
                EndIf
                &ErrorItem.Message = &Message.Description
                &ErrorResponse.Messages.Add(&ErrorItem)
                If &ConcatenatedMessage.IsEmpty()
                    &ConcatenatedMessage = &Message.Description
                Else
                    &ConcatenatedMessage = &ConcatenatedMessage + !" | " + &Message.Description
                EndIf
            EndIf
        EndFor
        If &ConcatenatedMessage.IsEmpty()
            &ErrorResponse.Message = !"Business rules rejected the request."
        Else
            If Len(&ConcatenatedMessage) > 2048
                &ErrorResponse.Message = Substr(&ConcatenatedMessage, 1, 2045) + !"..."
            Else
                &ErrorResponse.Message = &ConcatenatedMessage
            EndIf
        EndIf
    EndIf
EndIf
